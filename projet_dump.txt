  php:
    container_name: owner_php
    build:
      context: .
      target: app_php
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "php", "-v"]
      interval: 10s
      timeout: 3s
      retries: 3
      start_period: 30s
    volumes:
      - php_socket:/var/run/php
      - storage_data:/srv/app/var/storage
    environment:
      APP_ENV: prod
      APP_SECRET: ${APP_SECRET}
      DATABASE_URL: ${DATABASE_URL}
      MESSENGER_TRANSPORT_DSN: ${MESSENGER_TRANSPORT_DSN}
      MAILER_DSN: ${MAILER_DSN}
    depends_on:
      database:
        condition: service_healthy
    networks: [owner]

  worker:
    container_name: owner_worker
    build:
      context: .
      target: app_php
    restart: unless-stopped
    command: php bin/console messenger:consume async --time-limit=3600 -vv
    volumes:
      - storage_data:/srv/app/var/storage
    environment:
      APP_ENV: prod
      APP_SECRET: ${APP_SECRET}
      DATABASE_URL: ${DATABASE_URL}
      MESSENGER_TRANSPORT_DSN: ${MESSENGER_TRANSPORT_DSN}
      MAILER_DSN: ${MAILER_DSN}
    depends_on:
      database:
        condition: service_healthy
    networks: [owner]

  caddy:
    container_name: owner_caddy
    build:
      context: .
      target: app_caddy
    restart: unless-stopped
    depends_on: [php]
    environment:
      SERVER_NAME: ${SERVER_NAME:-localhost, caddy:80}
    ports:
      - "${HTTP_PORT:-80}:80"
      - "${HTTPS_PORT:-443}:443"
      - "${HTTPS_PORT:-443}:443/udp"
    volumes:
      - php_socket:/var/run/php
      - caddy_data:/data
      - caddy_config:/config
      - storage_data:/srv/app/var/storage:ro
    networks: [owner]

  database:
    container_name: owner_db
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_DB: ${POSTGRES_DB:-owner}
      POSTGRES_USER: ${POSTGRES_USER:-owner}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-owner}
    volumes:
      - db_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-owner}"]
      interval: 5s
      timeout: 3s
      retries: 10
    networks: [owner]

----- Dockerfile -----
#syntax=docker/dockerfile:1.4

ARG PHP_VERSION=8.3
ARG CADDY_VERSION=2.8

# ==========================================================
# Stage PHP-FPM (base prod)
# ==========================================================
FROM php:${PHP_VERSION}-fpm-alpine AS app_php

ENV APP_ENV=prod

WORKDIR /srv/app

RUN apk add --no-cache \
        acl file gettext git make bash fcgi \
        icu-dev libzip-dev postgresql-dev oniguruma-dev linux-headers $PHPIZE_DEPS \
    && docker-php-ext-install -j$(nproc) intl zip pdo pdo_pgsql opcache \
    && pecl install apcu && docker-php-ext-enable apcu \
    && apk del $PHPIZE_DEPS \
    && rm -rf /var/cache/apk/*

COPY --from=composer/composer:2-bin /composer /usr/bin/composer
ENV COMPOSER_ALLOW_SUPERUSER=1

COPY docker/php/conf.d/app.ini $PHP_INI_DIR/conf.d/
COPY docker/php/conf.d/app.prod.ini $PHP_INI_DIR/conf.d/
COPY docker/php/php-fpm.d/zz-docker.conf /usr/local/etc/php-fpm.d/zz-docker.conf
RUN mkdir -p /var/run/php

COPY docker/php/docker-entrypoint.sh /usr/local/bin/docker-entrypoint
RUN chmod +x /usr/local/bin/docker-entrypoint

COPY composer.json composer.lock symfony.lock ./
RUN composer install --prefer-dist --no-dev --no-autoloader --no-scripts --no-progress \
    && composer clear-cache

COPY . .
RUN rm -rf docker/ \
    && mkdir -p var/cache var/log var/storage \
    && composer dump-autoload --classmap-authoritative --no-dev \
    && composer dump-env prod \
    && php bin/console cache:warmup \
    && chmod +x bin/console

ENTRYPOINT ["docker-entrypoint"]
CMD ["php-fpm"]

# ==========================================================
# Stage DEV
# ==========================================================
FROM app_php AS app_php_dev

ENV APP_ENV=dev XDEBUG_MODE=off

RUN rm $PHP_INI_DIR/conf.d/app.prod.ini \
    && mv "$PHP_INI_DIR/php.ini" "$PHP_INI_DIR/php.ini-production" \
    && mv "$PHP_INI_DIR/php.ini-development" "$PHP_INI_DIR/php.ini" \
    && apk add --no-cache --virtual .build-deps $PHPIZE_DEPS linux-headers \
    && pecl install xdebug && docker-php-ext-enable xdebug \
    && apk del .build-deps \
    && rm -f .env.local.php

COPY docker/php/conf.d/app.dev.ini $PHP_INI_DIR/conf.d/

# ==========================================================
# Stage Caddy (sans Mercure/Vulcain)
# ==========================================================
FROM caddy:${CADDY_VERSION}-alpine AS app_caddy

WORKDIR /srv/app

COPY docker/caddy/Caddyfile /etc/caddy/Caddyfile
COPY --from=app_php /srv/app/public public/
